import React, { useState } from 'react';
import { Plus, ChevronDown, HelpCircle } from 'lucide-react';
import { Button } from '@/components/ui/button';
import { ResizeHandle } from '../shared/ResizeHandle';
import AddResourceDropdown, { ResourceType } from './AddResourceDropdown';
import ResourceDisplay from './ResourceDisplay';

const typeColors: { [key in ResourceType]: string } = {
  cuecard: 'bg-white border-gray-200 hover:bg-gray-50',
  note: 'bg-white border-gray-200 hover:bg-gray-50',
  image: 'bg-white border-gray-200 hover:bg-gray-50',
  quiz: 'bg-white border-gray-200 hover:bg-gray-50',
};

export interface CardDisplayProps {
  width: number;
  isEditMode: boolean;
  currentTopicTitle: string; // New prop to receive the current topic title
  onEditToggle: () => void;
  onResize: (width: number) => void;
}

export const CardDisplay: React.FC<CardDisplayProps> = ({
  width,
  isEditMode,
  currentTopicTitle, // Add to props destructuring
  onEditToggle,
  onResize,
}) => {
  // Update resource state interface to include topicTitle
  const [resources, setResources] = useState<
    {
      id: string;
      type: ResourceType;
      topicTitle: string;
      title: string;
      content: string;
      isEditing?: boolean;
    }[]
  >([]);

  const [newResourceType, setNewResourceType] = useState<ResourceType | null>(null);

  const handleAddResource = (type: ResourceType) => {
    const newResource = {
      id: Date.now().toString(),
      type,
      topicTitle: currentTopicTitle, // Use the current topic title for new resources
      title: '',
      content: '',
      isEditing: true,
    };
    setResources([...resources, newResource]);
    setNewResourceType(null);
  };

  const handleSave = (id: string, updatedData: Partial<typeof resources[number]>) => {
    setResources(
      resources.map((resource) =>
        resource.id === id 
          ? { 
              ...resource, 
              ...updatedData,
              topicTitle: currentTopicTitle // Always use current topic title when saving
            } 
          : resource
      )
    );
  };

  const handleDelete = (id: string) => {
    setResources(resources.filter((resource) => resource.id !== id));
  };

  const handleAskAI = (resource: typeof resources[number]) => {
    console.log('Ask AI about:', resource);
  };

  const handleImageCaption = (resource: typeof resources[number]) => {
    console.log('Generate caption for:', resource);
  };

  return (
    <div className="relative h-full" style={{ width }}>
      <div className="h-full bg-[#F8F9FA] border-l flex flex-col">
        {/* Header section */}
        <div className="p-4 border-b flex justify-between items-center bg-[#F8F9FA] sticky top-0 z-10">
          <h2 className="font-semibold text-xl text-gray-800">Resources</h2>
          <div className="relative">
            <AddResourceDropdown
              newResourceType={newResourceType}
              onSelectType={handleAddResource}
            />
          </div>
        </div>

        {/* Main content area */}
        <div className="flex-1 overflow-y-auto p-4 space-y-4">
          {resources.map((resource) => (
            <ResourceDisplay
              key={resource.id}
              id={resource.id}
              type={resource.type}
              topicTitle={resource.topicTitle} // Pass the topic title
              title={resource.title}
              content={resource.content}
              isEditing={resource.isEditing ?? false}
              typeColors={typeColors}
              onTitleChange={(title) => handleSave(resource.id, { title })}
              onContentChange={(content) => handleSave(resource.id, { content })}
              onCancel={() => handleSave(resource.id, { isEditing: false })}
              onSave={() => handleSave(resource.id, { isEditing: false })}
              onEdit={() => handleSave(resource.id, { isEditing: true })}
              onDelete={() => handleDelete(resource.id)}
              onAskAI={() => handleAskAI(resource)}
              onImageCaption={() => handleImageCaption(resource)}
            />
          ))}
        </div>

        {/* Footer section */}
        <div className="border-t p-4 bg-gray-50 sticky bottom-0">
          <Button
            variant="outline"
            className="w-full bg-[#F8F9FA] hover:bg-gradient-to-r hover:from-sky-500 hover:to-indigo-500 hover:text-white transition-all duration-300 text-base py-2"
            onClick={() => {
              /* Handle AI interaction */
            }}
          >
            <HelpCircle className="h-4 w-4 mr-2" />
            Ask AI Assistant
          </Button>
        </div>
      </div>
      <ResizeHandle placement="left" onDrag={onResize} />
    </div>
  );
};

export default CardDisplay;
