// authRoutes.ts

/**
 * authRoutes.ts
 * 
 * Provides authentication-related API routes using Express Router.
 * Responsibilities:
 * - Defines routes for user authentication via Google OAuth2.
 * - Implements a silent callback endpoint (`/oauth/callback`) for OAuth2 token exchanges.
 * - Provides a login route for direct user authentication.
 * - Communicates tokens securely to the parent app using `postMessage`.
 * Dependencies:
 * - Express for routing.
 * - AuthService for handling token exchanges and Google OAuth2 logic.
 * - AuthController for handling authentication requests.
 * - RateLimiter to prevent API abuse.
 */

import express from 'express';
import AuthService from '../services/authService';
import AuthController from '../controllers/authController';
import RateLimiter from '../middleware/rateLimiter';

const router = express.Router();

/**
 * POST /api/login
 * 
 * Authenticates a user using a Google OAuth token and returns a JWT.
 * Rate-limited to prevent abuse.
 */
router.post('/login', RateLimiter, AuthController.login);
router.post('/drive/save', AuthController.saveFile);
router.get('/drive/load/:fileId', AuthController.loadFile);
router.get('/drive/list', AuthController.listFiles);
router.post('/drive/share', AuthController.shareFile);
router.put('/drive/update', AuthController.updateFile);

/**
 * GET /oauth/callback
 * 
 * Silent OAuth2 callback endpoint.
 * Handles the token exchange after Google redirects the user here and uses `postMessage`
 * to securely send tokens and state back to the parent application.
 * 
 * Query Parameters:
 * - code: Authorization code from Google.
 * - state: Encoded user intent (e.g., `file-sharing`, `premium-service`).
 * 
 * Responses:
 * - Sends an HTML script to post the tokens and state back to the parent window and closes the popup.
 */
router.get('/oauth/callback', async (req, res) => {
  const { code, state } = req.query;

  try {
    if (!code) {
      throw new Error('Authorization code is missing.');
    }

    // Exchange the authorization code for tokens
    const tokens = await AuthService.getTokensFromCode(code as string);

    // Generate the HTML response to send tokens and state back to the parent app
    const responseHtml = `
      <script>
        // Post tokens and state back to the parent window
        window.opener.postMessage(
          ${JSON.stringify({ tokens, state })},
          window.opener.location.origin
        );
        // Close the popup
        window.close();
      </script>
    `;
    res.send(responseHtml);
  } catch (error) {
    const errorMessage = error instanceof Error ? error.message : 'Unknown error occurred';
    console.error('OAuth2 callback error:', errorMessage);
    res.status(400).send('Authentication failed.');
  }
});

export default router;

import { Request, Response } from 'express';
import AuthService from '../services/authService';
import DriveFileManager from '../services/DriveFileManager';

/**
 * AuthController
 * 
 * Contains methods to handle authentication-related HTTP requests.
 */
class AuthController {
  /**
   * Handles user login via Google OAuth token.
   * 
   * @param req - Express Request object containing the Google OAuth token.
   * @param res - Express Response object to send back the JWT and user details.
   */
  public async login(req: Request, res: Response): Promise<void> {
    try {
      const { token } = req.body;
      if (!token) {
        res.status(400).json({ status: 'error', message: 'Token is required.' });
        return;
      }

      // Validate the Google OAuth token
      const googleUserData = await AuthService.validateToken(token);
      if (!googleUserData) {
        res.status(401).json({ status: 'error', message: 'Invalid token.' });
        return;
      }

      // Retrieve or create the user
      const user = await AuthService.retrieveUser(googleUserData);

      // Generate JWT
      const jwtToken = AuthService.generateJWT(user);

      res.status(200).json({
        status: 'success',
        token: jwtToken,
        user: {
          email: user.email,
          accountType: user.accountType,
        },
      });
    } catch (error) {
      console.error('Login error:', error);
      res.status(500).json({ status: 'error', message: 'Internal server error.' });
    }
  }
  public async saveFile(req: Request, res: Response): Promise<void> {
    try {
      const { fileName, content, folderId } = req.body;
      const accessToken = req.headers.authorization?.split(' ')[1];

      if (!accessToken) {
        res.status(401).json({ status: 'error', message: 'No access token provided' });
        return;
      }

      await DriveFileManager.initialize(accessToken);
      const fileId = await DriveFileManager.saveToGoogleDrive(fileName, Buffer.from(content), folderId);

      res.status(200).json({
        status: 'success',
        fileId,
      });
    } catch (error) {
      console.error('Save file error:', error);
      res.status(500).json({ status: 'error', message: 'Failed to save file' });
    }
  }

  public async loadFile(req: Request, res: Response): Promise<void> {
    try {
      const { fileId } = req.params;
      const accessToken = req.headers.authorization?.split(' ')[1];

      if (!accessToken) {
        res.status(401).json({ status: 'error', message: 'No access token provided' });
        return;
      }

      await DriveFileManager.initialize(accessToken);
      const content = await DriveFileManager.loadFromGoogleDrive(fileId);

      res.status(200).json({
        status: 'success',
        content: content.toString('base64'),
      });
    } catch (error) {
      console.error('Load file error:', error);
      res.status(500).json({ status: 'error', message: 'Failed to load file' });
    }
  }

  public async listFiles(req: Request, res: Response): Promise<void> {
    try {
      const accessToken = req.headers.authorization?.split(' ')[1];

      if (!accessToken) {

  }

      const { fileId, settings } = req.body;
      const accessToken = req.headers.authorization?.split(' ')[1];

      if (!accessToken) {
        res.status(401).json({ status: 'error', message: 'No access token provided' });

      console.error('Share file error:', error);

const API_BASE_URL = process.env.NEXT_PUBLIC_API_URL || 'http://localhost:3000';

export class FrontendDriveService {
  private static getHeaders(accessToken: string) {
    return {
      'Authorization': `Bearer ${accessToken}`,
      'Content-Type': 'application/json'
    };
  }

  static async saveFile(fileName: string, content: any, accessToken: string, folderId?: string) {
    try {
      const response = await fetch(`${API_BASE_URL}/api/drive/save`, {
        method: 'POST',
        headers: this.getHeaders(accessToken),
        body: JSON.stringify({ fileName, content, folderId })
      });

      if (!response.ok) {
        throw new Error('Failed to save file');
      }

      const data = await response.json();
      return data.fileId;
    } catch (error) {
      console.error('Error saving file:', error);
      throw error;
    }
  }

  static async loadFile(fileId: string, accessToken: string) {
    try {
      const response = await fetch(`${API_BASE_URL}/api/drive/load/${fileId}`, {
        method: 'GET',
        headers: this.getHeaders(accessToken)
      });

      if (!response.ok) {
        throw new Error('Failed to load file');
      }

      const data = await response.json();
      return Buffer.from(data.content, 'base64');
    } catch (error) {
      console.error('Error loading file:', error);
      throw error;
    }
  }

  static async listFiles(accessToken: string) {
    try {
      const response = await fetch(`${API_BASE_URL}/api/drive/list`, {
        method: 'GET',
        headers: this.getHeaders(accessToken)
      });

      if (!response.ok) {
        throw new Error('Failed to list files');
      }

      const data = await response.json();
      return data.files;
    } catch (error) {
      console.error('Error listing files:', error);
      throw error;
    }
  }

  static async shareFile(fileId: string, settings: any, accessToken: string) {
    try {
      const response = await fetch(`${API_BASE_URL}/api/drive/share`, {
        method: 'POST',
        headers: this.getHeaders(accessToken),
        body: JSON.stringify({ fileId, settings })
      });

      if (!response.ok) {
        throw new Error('Failed to share file');
      }

      return await response.json();
    } catch (error) {
      console.error('Error sharing file:', error);
      throw error;
    }
  }

  static async updateFile(fileId: string, content: any, accessToken: string) {
    try {
      const response = await fetch(`${API_BASE_URL}/api/drive/update`, {
        method: 'PUT',
        headers: this.getHeaders(accessToken),
        body: JSON.stringify({ fileId, content })
      });

      if (!response.ok) {
        throw new Error('Failed to update file');
      }

      return await response.json();
    } catch (error) {
      console.error('Error updating file:', error);
      throw error;
    }
  }
}

  visible,
  onClose,

      setLoading(false);

      setLoading(true);

        type: 'user',

import React, { createContext, useState, useEffect } from 'react';

interface User {
  email: string;
  accountType: 'free' | 'premium';
  accessToken: string;
}

interface AuthContextType {
  isAuthenticated: boolean;
  isGuest: boolean;
  user: User | null;
  signIn: (token: string) => void;
  signOut: () => void;
  skipAuth: () => void;
}

export const AuthContext = createContext<AuthContextType>({
  isAuthenticated: false,
  isGuest: false,
  user: null,
  signIn: () => {},
  signOut: () => {},
  skipAuth: () => {},
});


export const AuthProvider: React.FC<React.PropsWithChildren<{}>> = ({ children }) => {
  const [user, setUser] = useState<User | null>(null);
  const [isGuest, setIsGuest] = useState<boolean>(false);

  useEffect(() => {
    const token = localStorage.getItem('token');
    const guest = localStorage.getItem('isGuest') === 'true';
    if (token) {
      // Optionally verify token with backend
      const decoded = JSON.parse(atob(token.split('.')[1]));
      setUser({ accessToken: token, email: decoded.email, accountType: decoded.accountType });
    }
    if (guest) {
      setIsGuest(true);
    }
  }, []);

  const signIn = (token: string) => {
    localStorage.setItem('token', token);
    // Decode token to extract user info
    const decoded = JSON.parse(atob(token.split('.')[1]));
    setUser({ accessToken: token, email: decoded.email, accountType: decoded.accountType });
    setIsGuest(false);
  };

  const signOut = () => {
    localStorage.removeItem('token');
    setUser(null);
    setIsGuest(false);
  };

  const skipAuth = () => {
    localStorage.setItem('isGuest', 'true');
    setIsGuest(true);
    setUser(null);
  };

  return (
    <AuthContext.Provider value={{ isAuthenticated: !!user, isGuest, user, signIn, signOut, skipAuth }}>
      {children}
    </AuthContext.Provider>
  );
};

    try {
      const content = await DriveService.getFile(fileId);
      return content;
    } catch (error) {
      console.error('Error loading from Google Drive:', error);
      throw new Error('Failed to load file from Google Drive');
    }

   * Lists available .bloomdocx files in Google Drive
   * @returns Promise resolving to an array of DriveFile objects
   */
  async listDriveFiles(): Promise<DriveFile[]> {
    try {
      const files = await DriveService.listFiles();
      return files;
    } catch (error) {
      console.error('Error listing Drive files:', error);
      throw new Error('Failed to list Drive files');
    }
  }

   * Configures autosave functionality
   * @param config - AutoSave configuration options
   */
  configureAutosave(config: AutoSaveConfig): void {
    AutoSaveService.configure(config);
  }

  /**
   * Shares a file with specified settings
   * @param fileId - ID of the file to share
   * @param settings - Share settings including permissions and target users
   */
  async shareFile(fileId: string, settings: ShareSettings): Promise<void> {

   * @param content - New content for the file
   */
  async updateFile(fileId: string, content: Buffer): Promise<void> {
    try {
      await DriveService.updateFile(fileId, content);
    } catch (error) {
      console.error('Error updating file:', error);
      throw new Error('Failed to update file');
    }
  }
}
