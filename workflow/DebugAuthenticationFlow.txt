/**
 * GET /oauth/callback
 * 
 * Silent OAuth2 callback endpoint.
 * Handles the token exchange after Google redirects the user here and uses `postMessage`
 * to securely send tokens and state back to the parent application.
 * 
 * Query Parameters:
 * - code: Authorization code from Google.
 * - state: Encoded user intent (e.g., `file-sharing`, `premium-service`).
 * 
 * Responses:
 * - Sends an HTML script to post the tokens and state back to the parent window and closes the popup.
 */
router.get('/oauth/callback', async (req, res) => {
  const { code, state } = req.query;

  try {
    if (!code) {
      throw new Error('Authorization code is missing.');
    }

    // Exchange the authorization code for tokens
    const tokens = await AuthService.getTokensFromCode(code as string);

    // Generate the HTML response to send tokens and state back to the parent app
    const responseHtml = `
      <script>
        // Post tokens and state back to the parent window
        window.opener.postMessage(
          ${JSON.stringify({ tokens, state })},
          window.opener.location.origin
        );
        // Close the popup
        window.close();
      </script>
    `;
    res.send(responseHtml);
  } catch (error) {
    const errorMessage = error instanceof Error ? error.message : 'Unknown error occurred';
    console.error('OAuth2 callback error:', errorMessage);
    res.status(400).send('Authentication failed.');
  }
});

/**
 * authService.ts
 * 
 * Provides a complete OAuth2 flow for Google authentication and API access.
 * Responsibilities:
 * - Handles Google OAuth2 authentication for login and API actions.
 * - Manages tokens (access and refresh) for extended user sessions.
 * - Generates signed JWTs for application-level authentication.
 * Dependencies:
 * - google-auth-library for OAuth2 flow.

import { OAuth2Client } from 'google-auth-library';
import UserModel, { IUser } from '../models/userModel';
import JwtUtils from '../utils/jwtUtils';
import DriveService from './driveService';

class AuthService {
  private googleClient: OAuth2Client;

  constructor() {
    this.googleClient = new OAuth2Client(
      process.env.GOOGLE_CLIENT_ID,
      process.env.GOOGLE_CLIENT_SECRET,
      process.env.REDIRECT_URL // Ensure these are set in the environment variables
    );
  }

  /**
   * Generates the authorization URL for Google OAuth2.
   * Includes the `state` parameter to track the user's intent (e.g., `file-sharing`, `premium-service`).
   * @param actionType - The user's intended action after authentication.
   * @returns The Google OAuth2 authorization URL.
   */
  generateAuthUrl(actionType: string): string {
    return this.googleClient.generateAuthUrl({
      access_type: 'offline', // Required for refresh tokens
      scope: [
        'https://www.googleapis.com/auth/userinfo.profile',
        'https://www.googleapis.com/auth/drive.file', // Access to user's Google Drive
        'https://www.googleapis.com/auth/drive.appdata',
      ],
      prompt: 'consent', // Ensures refresh tokens are provided
      state: actionType, // Encodes the user's intent
    });
  }

  /**
   * Exchanges the authorization code for access and refresh tokens.
   * @param code - The authorization code from the user's consent.
   * @returns The tokens (access and refresh) provided by Google.
   */
  async getTokensFromCode(code: string): Promise<any> {
    const { tokens } = await this.googleClient.getToken(code);
    this.googleClient.setCredentials(tokens);
    return tokens;
  }

  /**
   * Processes the `state` parameter to determine the post-authentication action.
   * @param state - The user's intended action (`file-sharing`, `premium-service`, etc.).
   * @param tokens - Tokens provided by Google after authentication.
   */
  async handlePostAuthAction(state: string, tokens: any): Promise<void> {
    // Save tokens securely or pass to the appropriate service
    await DriveService.initializeDriveClient(tokens.access_token);
    if (state === 'file-sharing') {
      // Example: initiate a file-sharing action
      console.log('Initiate file-sharing with tokens:', tokens);
    } else if (state === 'premium-service') {
      // Example: enable premium features
      console.log('Enable premium services with tokens:', tokens);
    }
  }

  /**
   * Validates an existing Google OAuth token.
   * @param token - The Google OAuth token to validate.
   * @returns Decoded token data if valid.
   */
  async validateToken(token: string): Promise<any> {
    try {
      // For access tokens, we need to call userinfo endpoint
      const response = await fetch(
        'https://www.googleapis.com/oauth2/v3/userinfo',
        {
          headers: {
            Authorization: `Bearer ${token}`,
          },
        }
      );
      
      if (!response.ok) {
        throw new Error('Failed to validate token');
      }
      
      const userData = await response.json();
      return {
        sub: userData.sub,
        email: userData.email,
        // other user info you need
      };
    } catch (error) {
      throw new Error('Invalid Google OAuth token.');
    }
  }


  /**
   * Retrieves or creates a user in the database based on Google profile data.
   * @param googleUserData - Data from the validated Google token.
   * @returns The user object from the database.
   */
  async retrieveUser(googleUserData: any): Promise<IUser> {
    const { sub: googleId, email } = googleUserData;
    let user = await UserModel.findOne({ googleId });
    if (!user) {
      user = await UserModel.create({ googleId, email });
    }
    return user;
  }

  /**
   * Generates a JWT for application-level authentication.
   * @param user - The authenticated user object.
   * @returns A signed JWT containing user details.
   */
  generateJWT(user: IUser): string {
    return JwtUtils.generateJWT({
      id: user._id,
      email: user.email,
      accountType: user.accountType,
    });
  }
}

/**
 * src/services/driveService.ts

 * Provides methods to interact with Google Drive API for file operations.
 * Handles file saving, retrieval, and updates for .bloomdocx files.
 * 
 * Dependencies:
 * - google-auth-library for OAuth2
 * - googleapis for Drive API
 * - AuthService for token management
 * - Environment variables for API configuration
 */

import { drive_v3, google } from 'googleapis';
import { OAuth2Client } from 'google-auth-library';


export interface DriveFile {
  id: string;
  name: string;
  mimeType: string;
  modifiedTime?: string;
}

export class DriveService {
  private static instance: DriveService;
  private drive: drive_v3.Drive | null = null;
  private accessToken: string | null = null;
  private constructor() {}

  static getInstance(): DriveService {
    if (!DriveService.instance) {
      DriveService.instance = new DriveService();
    }
    return DriveService.instance;
  }

  getDriveInstance(): drive_v3.Drive {
    if (!this.drive) throw new Error('Drive client not initialized');
    return this.drive;
  }

  async initializeDriveClient(accessToken: string): Promise<void> {
    console.log('Access token received in driveService', accessToken);
    
    try {
      // Verify the access token by making a test request to userinfo endpoint
      const response = await fetch('https://www.googleapis.com/oauth2/v3/userinfo', {
        headers: {
          'Authorization': `Bearer ${accessToken}`
        }
      });

      if (!response.ok) {
        throw new Error('Invalid access token');
      }

      // If we get here, the token is valid
      this.accessToken = accessToken;
      
      // Create OAuth2Client with the access token
      const auth = new OAuth2Client();
      auth.setCredentials({
        access_token: accessToken
      });
      
      // Initialize the Drive client
      this.drive = google.drive({ 
        version: 'v3', 
        auth 
      });
      
    } catch (error) {
      console.error('Failed to initialize Drive client:', error);
      throw error;
    }
  }
  private async fetchWithAuth(endpoint: string, options: RequestInit = {}): Promise<Response> {
    if (!this.accessToken) {
      throw new Error('Drive client not initialized');
    }

    try {
      const baseUrl = 'https://www.googleapis.com/drive/v3';
      const headers = new Headers(options.headers);
      headers.set('Authorization', `Bearer ${this.accessToken}`);
      
      if (!headers.has('Content-Type')) {
        headers.set('Content-Type', 'application/json');
      }
  
      const response = await fetch(`${baseUrl}${endpoint}`, {
        ...options,
        headers
      });
  
      if (!response.ok) {
        const errorData = await response.json().catch(() => null);
        throw new Error(
          `Drive API error: ${response.status} ${JSON.stringify(errorData, null, 2)}`
        );
      }
  
      return response;
    } catch (error) {
      console.error('API request failed:', error);
      throw error;
    }
  }

  async saveFile(fileName: string, content: Buffer, folderId?: string): Promise<string> {
    const metadata = {
      name: `${fileName}.bloomdocx`,
      mimeType: 'application/vnd.bloom.document',
      ...(folderId && { parents: [folderId] }),
    };

    // Create multipart form-data
    const formData = new FormData();
    formData.append(
      'metadata',
      new Blob([JSON.stringify(metadata)], { type: 'application/json' })
    );
    formData.append(
      'file',
      new Blob([content], { type: 'application/octet-stream' })
    );

    const response = await this.fetchWithAuth('/files?uploadType=multipart', {
      method: 'POST',
      body: formData,
      headers: {
        // Don't set Content-Type here - browser will set it with boundary
      }
    });

    const data = await response.json();
    return data.id;
  }


  async updateFile(fileId: string, content: Buffer): Promise<void> {
    await this.fetchWithAuth(`/files/${fileId}`, {
      method: 'PATCH',
      body: content,
      headers: {
        'Content-Type': 'application/octet-stream'
      }
    });
  }

  async getFile(fileId: string): Promise<Buffer> {
    const response = await this.fetchWithAuth(`/files/${fileId}?alt=media`);
    const arrayBuffer = await response.arrayBuffer();
    return Buffer.from(arrayBuffer);
  }

  async listFiles(): Promise<DriveFile[]> {
    const response = await this.fetchWithAuth('/files', {
      method: 'GET',
    });
    
    const data = await response.json();
    return data.files || [];
  }
}
