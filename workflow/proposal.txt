// 2. This Utility save method from Document utils is used to zip and process the file into a .bloomdocx,
this file is essentially a set of jsonified strings that are stored in a zip folder. After zipping the file,
the file is sent to the back end using the predefined request in the frontendDriveService file.

export const saveToGoogleDrive = async (
  nodes: Map<string, ConceptNode>,
  colorTheme: ColorTheme,
  primaryColor: string,
  backgroundColor: string,
  fileName: string,
  cueCards: Map<string, ResourceCard[]>,
  subtrees: Map<string, SubtreeData>,
  pictures: Map<string, ResourceCard[]>,
  practiceSheets: Map<string, ResourceCard[]>,
  accessToken: string
): Promise<string> => {
  const zip = new JSZip();

  // Reuse existing serialization logic
  const serializedDocument = serializeDocument(
    nodes,
    colorTheme,
    primaryColor,
    backgroundColor,
    fileName
  );
  const documentJson = JSON.stringify(serializedDocument);
  zip.file('document/document.json', documentJson);

  // Add other files using existing serialization methods
  const serializedCueCards = serializeCueCards(cueCards as Map<string, CueCardResource[]>);
  zip.file('document/cuecards.json', JSON.stringify(serializedCueCards));

  const serializedSubtrees = serializeSubtrees(subtrees);
  zip.file('document/subtrees.json', JSON.stringify(serializedSubtrees));

  const serializedPictures = serializePictures(pictures as Map<string, PictureResource[]>);
  zip.file('document/pictures.json', JSON.stringify(serializedPictures));

  const serializedPracticeSheets = serializePracticeSheets(
    practiceSheets as Map<string, PracticeSheetResource[]>
  );
  zip.file('document/practiceSheets.json', JSON.stringify(serializedPracticeSheets));

  // Add images to the ZIP
  for (const [nodeId, pictureArray] of pictures.entries()) {
    for (const picture of pictureArray as PictureResource[]) {
      if (picture.image) {
        try {
          const imageData = await picture.image.arrayBuffer();
          zip.file(`document/images/${picture.id}_${picture.image.name}`, imageData);
        } catch (error) {
          console.error(`Error saving image ${picture.id}:`, error);
        }
      }
    }
  }

  // Generate ZIP content as Buffer
  const content = await zip.generateAsync({ type: 'arraybuffer' });
  
  // Save to Google Drive
  return  FrontendDriveService.saveFile(fileName, Buffer.from(content), accessToken);
};

const API_BASE_URL = process.env.NEXT_PUBLIC_API_SERVER_URL || 'http://localhost:5000';

export class FrontendDriveService {
  private static getHeaders(accessToken: string) {
    return {
      'Authorization': `Bearer ${accessToken}`,
      'Content-Type': 'application/json'
    };
  }

  static async saveFile(fileName: string, content: any, accessToken: string, folderId?: string) {
    try {
      const response = await fetch(`${API_BASE_URL}/api/drive/save`, {
        method: 'POST',
        headers: this.getHeaders(accessToken),
        body: JSON.stringify({ fileName, content, folderId })
      });

      if (!response.ok) {
        throw new Error('Failed to save file');
      }

      const data = await response.json();
      return data.fileId;
    } catch (error) {
      console.error('Error saving file:', error);
      throw error;
    }
  }

  static async loadFile(fileId: string, accessToken: string) {
    try {
      const response = await fetch(`${API_BASE_URL}/api/drive/load/${fileId}`, {
        method: 'GET',
        headers: this.getHeaders(accessToken)
      });

      if (!response.ok) {
        throw new Error('Failed to load file');
      }

      const data = await response.json();
      return Buffer.from(data.content, 'base64');
    } catch (error) {
      console.error('Error loading file:', error);
      throw error;
    }
  }

  static async listFiles(accessToken: string) {
    try {
      const response = await fetch(`${API_BASE_URL}/api/drive/list`, {
        method: 'GET',
        headers: this.getHeaders(accessToken)
      });

      if (!response.ok) {
        throw new Error('Failed to list files');
      }

      const data = await response.json();
      return data.files;
    } catch (error) {
      console.error('Error listing files:', error);
      throw error;
    }
  }

  static async shareFile(fileId: string, settings: any, accessToken: string) {
    try {
      const response = await fetch(`${API_BASE_URL}/api/drive/share`, {
        method: 'POST',
        headers: this.getHeaders(accessToken),
        body: JSON.stringify({ fileId, settings })
      });

      if (!response.ok) {
        throw new Error('Failed to share file');
      }

      return await response.json();
    } catch (error) {
      console.error('Error sharing file:', error);
      throw error;
    }
  }

  static async updateFile(fileId: string, content: any, accessToken: string) {
    try {
      const response = await fetch(`${API_BASE_URL}/api/drive/update`, {
        method: 'PUT',
        headers: this.getHeaders(accessToken),
        body: JSON.stringify({ fileId, content })
      });

      if (!response.ok) {
        throw new Error('Failed to update file');
      }

      return await response.json();
    } catch (error) {
      console.error('Error updating file:', error);
      throw error;
    }
  }
}

// driveService.ts

/**
 * src/services/driveService.ts
 * 
 * Provides methods to interact with Google Drive API for file operations.
 * Handles file saving, retrieval, and updates for .bloomdocx files.
 * 
 * Dependencies:
 * - google-auth-library for OAuth2
 * - googleapis for Drive API
 * - AuthService for token management
 * - Environment variables for API configuration
 */

import { drive_v3, google } from 'googleapis';
import { OAuth2Client } from 'google-auth-library';
import { Readable } from 'stream';

export interface DriveFile {
  id: string;
  name: string;
  mimeType: string;
  modifiedTime?: string;
}

export class DriveService {
  private static instance: DriveService;
  private drive: drive_v3.Drive | null = null;
  private accessToken: string | null = null;
  private constructor() {}

  static getInstance(): DriveService {
    if (!DriveService.instance) {
      DriveService.instance = new DriveService();
    }
    return DriveService.instance;
  }

  getDriveInstance(): drive_v3.Drive {
    if (!this.drive) throw new Error('Drive client not initialized');
    return this.drive;
  }

  async initializeDriveClient(accessToken: string): Promise<void> {
    try {
      const auth = new OAuth2Client({
        clientId: process.env.GOOGLE_CLIENT_ID,
        clientSecret: process.env.GOOGLE_CLIENT_SECRET,
      });
  
      auth.setCredentials({
        access_token: accessToken,
      });
  
      this.accessToken = accessToken;
      this.drive = google.drive({
        version: 'v3',
        auth,
      });
    } catch (error) {
      console.error('Failed to initialize Drive client:', error);
      throw error;
    }
  }
  
  private async fetchWithAuth(endpoint: string, options: RequestInit = {}): Promise<Response> {
    if (!this.accessToken) {
      throw new Error('Drive client not initialized');
    }

    try {
      const baseUrl = 'https://www.googleapis.com/drive/v3';
      const headers = new Headers(options.headers);
      headers.set('Authorization', `Bearer ${this.accessToken}`);
      
      if (!headers.has('Content-Type')) {
        headers.set('Content-Type', 'application/json');
      }
  
      const response = await fetch(`${baseUrl}${endpoint}`, {
        ...options,
        headers
      });
  
      if (!response.ok) {
        const errorData = await response.json().catch(() => null);
        throw new Error(
          `Drive API error: ${response.status} ${JSON.stringify(errorData, null, 2)}`
        );
      }
  
      return response;
    } catch (error) {
      console.error('API request failed:', error);
      throw error;
    }
  }

  async saveFile(fileName: string, content: Buffer, folderId?: string): Promise<string> {
    if (!this.drive) {
      throw new Error('Drive client not initialized');
    }

    try {
      // Create a readable stream from the buffer
      const fileStream = new Readable();
      fileStream.push(content);
      fileStream.push(null);

      const requestBody = {
        name: `${fileName}.bloomdocx`,
        mimeType: 'application/vnd.bloom.document',
        ...(folderId && { parents: [folderId] }),
      };

      const media = {
        mimeType: 'application/octet-stream',
        body: fileStream,
      };

      const response = await this.drive.files.create({
        requestBody,
        media,
        fields: 'id',
      });

      if (!response.data.id) {
        throw new Error('Failed to get file ID from response');
      }

      return response.data.id;
    } catch (error) {
      console.error('Error saving file to Drive:', error);
      throw error;
    }
  }


  async updateFile(fileId: string, content: Buffer): Promise<void> {
    if (!this.drive) {
      throw new Error('Drive client not initialized');
    }

    try {
      const fileStream = new Readable();
      fileStream.push(content);
      fileStream.push(null);

      await this.drive.files.update({
        fileId,
        media: {
          mimeType: 'application/octet-stream',
          body: fileStream,
        },
      });
    } catch (error) {
      console.error('Error updating file:', error);
      throw error;
    }
  }
  async getFile(fileId: string): Promise<Buffer> {
    if (!this.drive) {
      throw new Error('Drive client not initialized');
    }

    try {
      const response = await this.drive.files.get(
        { fileId, alt: 'media' },
        { responseType: 'arraybuffer' }
      );

      return Buffer.from(response.data as ArrayBuffer);
    } catch (error) {
      console.error('Error getting file:', error);
      throw error;
    }
  }

  async listFiles(): Promise<DriveFile[]> {
    if (!this.drive) {
      throw new Error('Drive client not initialized');
    }
  
    try {
      const response = await this.drive.files.list({
        q: "mimeType='application/vnd.bloom.document'",
        fields: 'files(id, name, mimeType, modifiedTime)',
      });
  
      const files = response.data.files || [];
      
      // Filter out invalid files and map to DriveFile
      return files
        .filter((file): file is Required<Pick<drive_v3.Schema$File, 'id' | 'name' | 'mimeType'>> & { modifiedTime?: string } =>
          !!file.id && !!file.name && !!file.mimeType
        )
        .map((file) => ({
          id: file.id!,
          name: file.name!,
          mimeType: file.mimeType!,
          modifiedTime: file.modifiedTime || undefined, // Ensure modifiedTime is optional
        }));
    } catch (error) {
      console.error('Error listing files:', error);
      throw error;
    }
  }
  
  
}

// context/AuthContext.tsx
import React, { createContext, useState, useEffect } from 'react';

interface User {
  email: string;
  accountType: 'free' | 'premium';
  accessToken: string;
  googleAccessToken: string;
}

interface AuthContextType {
  isAuthenticated: boolean;
  isGuest: boolean;
  user: User | null;
  signIn: (jwtToken: string, googleToken: string) => void;
  signOut: () => void;
  skipAuth: () => void;
}

export const AuthContext = createContext<AuthContextType>({
  isAuthenticated: false,
  isGuest: false,
  user: null,
  signIn: () => {},
  signOut: () => {},
  skipAuth: () => {},
});


export const AuthProvider: React.FC<React.PropsWithChildren<{}>> = ({ children }) => {
  const [user, setUser] = useState<User | null>(null);
  const [isGuest, setIsGuest] = useState<boolean>(false);

  useEffect(() => {
    const token = localStorage.getItem('token');
    const googleToken = localStorage.getItem('googleToken');
    const guest = localStorage.getItem('isGuest') === 'true';
    if (token && googleToken) {
      // Optionally verify token with backend
      const decoded = JSON.parse(atob(token.split('.')[1]));
      setUser({ accessToken: token, googleAccessToken: googleToken,
        email: decoded.email, accountType: decoded.accountType });
    }
    if (guest) {
      setIsGuest(true);
    }
  }, []);

  const signIn = (token: string, googleToken: string) => {
    localStorage.setItem('token', token);
    localStorage.setItem('googleToken', googleToken);
    // Decode token to extract user info
    const decoded = JSON.parse(atob(token.split('.')[1]));
    setUser({ accessToken: token, googleAccessToken: googleToken, email: decoded.email, accountType: decoded.accountType });
    setIsGuest(false);
  };

  const signOut = () => {
    localStorage.removeItem('token');
    localStorage.removeItem('googleToken');
    setUser(null);
    setIsGuest(false);
  };

  const skipAuth = () => {
    localStorage.setItem('isGuest', 'true');
    setIsGuest(true);
    setUser(null);
  };

  return (
    <AuthContext.Provider value={{ isAuthenticated: !!user, isGuest, user, signIn, signOut, skipAuth }}>
      {children}
    </AuthContext.Provider>
  );
};

/**
 * authRoutes.ts
 * 
 * Provides authentication-related API routes using Express Router.
 * Responsibilities:
 * - Defines routes for user authentication via Google OAuth2.
 * - Implements a silent callback endpoint (`/oauth/callback`) for OAuth2 token exchanges.
 * - Provides a login route for direct user authentication.
 * - Communicates tokens securely to the parent app using `postMessage`.
 * Dependencies:
 * - Express for routing.
 * - AuthService for handling token exchanges and Google OAuth2 logic.
 * - AuthController for handling authentication requests.
 * - RateLimiter to prevent API abuse.
 */

import express from 'express';
import AuthService from '../services/authService';
import AuthController from '../controllers/authController';
import RateLimiter from '../middleware/rateLimiter';

const router = express.Router();

/**
 * POST /api/login
 * 
 * Authenticates a user using a Google OAuth token and returns a JWT.
 * Rate-limited to prevent abuse.
 */
router.post('/login', RateLimiter, AuthController.login);
router.post('/drive/save', AuthController.saveFile);
router.get('/drive/load/:fileId', AuthController.loadFile);
router.get('/drive/list', AuthController.listFiles);
router.post('/drive/share', AuthController.shareFile);
router.put('/drive/update', AuthController.updateFile);

/**
 * GET /oauth/callback
 * 
 * Silent OAuth2 callback endpoint.
 * Handles the token exchange after Google redirects the user here and uses `postMessage`
 * to securely send tokens and state back to the parent application.
 * 
 * Query Parameters:
 * - code: Authorization code from Google.
 * - state: Encoded user intent (e.g., `file-sharing`, `premium-service`).
 * 
 * Responses:
 * - Sends an HTML script to post the tokens and state back to the parent window and closes the popup.
 */
router.get('/oauth/callback', async (req, res) => {
  const { code, state } = req.query;

  try {
    if (!code) {
      throw new Error('Authorization code is missing.');
    }

    // Exchange the authorization code for tokens
    const tokens = await AuthService.getTokensFromCode(code as string);

    // Generate the HTML response to send tokens and state back to the parent app
    const responseHtml = `
      <script>
        // Post tokens and state back to the parent window
        window.opener.postMessage(
          ${JSON.stringify({ tokens, state })},
          window.opener.location.origin
        );
        // Close the popup
        window.close();
      </script>
    `;
    res.send(responseHtml);
  } catch (error) {
    const errorMessage = error instanceof Error ? error.message : 'Unknown error occurred';
    console.error('OAuth2 callback error:', errorMessage);
    res.status(400).send('Authentication failed.');
  }
});

export default router;

/**
 * authController.ts
 * 
 * Handles API requests related to authentication, such as user login.
 */

import { Request, Response } from 'express';
import AuthService from '../services/authService';
import DriveFileManager from '../services/DriveFileManager';
import { ShareSettings } from '../services/driveShareService';

/**
 * AuthController
 * 
 * Contains methods to handle authentication-related HTTP requests.
 */
class AuthController {
  /**
   * Handles user login via Google OAuth token.
   * 
   * @param req - Express Request object containing the Google OAuth token.
   * @param res - Express Response object to send back the JWT and user details.
   */
  public async login(req: Request, res: Response): Promise<void> {
    try {
      const { token } = req.body;
      if (!token) {
        res.status(400).json({ status: 'error', message: 'Token is required.' });
        return;
      }

      // Validate the Google OAuth token
      const googleUserData = await AuthService.validateToken(token);
      if (!googleUserData) {
        res.status(401).json({ status: 'error', message: 'Invalid token.' });
        return;
      }

      // Retrieve or create the user
      const user = await AuthService.retrieveUser(googleUserData);

      // Generate JWT
      const jwtToken = AuthService.generateJWT(user);

      res.status(200).json({
        status: 'success',
        token: jwtToken,
        user: {
          email: user.email,
          accountType: user.accountType,
        },
      });
    } catch (error) {
      console.error('Login error:', error);
      res.status(500).json({ status: 'error', message: 'Internal server error.' });
    }
  }
  public async saveFile(req: Request, res: Response): Promise<void> {
    try {
      const { fileName, content, folderId } = req.body;
      const accessToken = req.headers.authorization?.split(' ')[1];

      if (!accessToken) {
        res.status(401).json({ status: 'error', message: 'No access token provided' });
        return;
      }

      await DriveFileManager.initialize(accessToken);
      const fileId = await DriveFileManager.saveToGoogleDrive(fileName, Buffer.from(content), folderId);

      res.status(200).json({
        status: 'success',
        fileId,
      });
    } catch (error) {
      console.error('Save file error:', error);
      res.status(500).json({ status: 'error', message: 'Failed to save file' });
    }
  }

  public async loadFile(req: Request, res: Response): Promise<void> {
    try {
      const { fileId } = req.params;
      const accessToken = req.headers.authorization?.split(' ')[1];

      if (!accessToken) {
        res.status(401).json({ status: 'error', message: 'No access token provided' });
        return;
      }

      await DriveFileManager.initialize(accessToken);
      const content = await DriveFileManager.loadFromGoogleDrive(fileId);

      res.status(200).json({
        status: 'success',
        content: content.toString('base64'),
      });
    } catch (error) {
      console.error('Load file error:', error);
      res.status(500).json({ status: 'error', message: 'Failed to load file' });
    }
  }

  public async listFiles(req: Request, res: Response): Promise<void> {
    try {
      const accessToken = req.headers.authorization?.split(' ')[1];

      if (!accessToken) {
        res.status(401).json({ status: 'error', message: 'No access token provided' });
        return;
      }

      await DriveFileManager.initialize(accessToken);
      const files = await DriveFileManager.listDriveFiles();

      res.status(200).json({
        status: 'success',
        files,
      });
    } catch (error) {
      console.error('List files error:', error);
      res.status(500).json({ status: 'error', message: 'Failed to list files' });
    }
  }

  public async shareFile(req: Request, res: Response): Promise<void> {
    try {
      const { fileId, settings } = req.body;
      const accessToken = req.headers.authorization?.split(' ')[1];

      if (!accessToken) {
        res.status(401).json({ status: 'error', message: 'No access token provided' });
        return;
      }

      await DriveFileManager.initialize(accessToken);
      await DriveFileManager.shareFile(fileId, settings as ShareSettings);

      res.status(200).json({
        status: 'success',
        message: 'File shared successfully',
      });
    } catch (error) {
      console.error('Share file error:', error);
      res.status(500).json({ status: 'error', message: 'Failed to share file' });
    }
  }

  public async updateFile(req: Request, res: Response): Promise<void> {
    try {
      const { fileId, content } = req.body;
      const accessToken = req.headers.authorization?.split(' ')[1];

      if (!accessToken) {
        res.status(401).json({ status: 'error', message: 'No access token provided' });
        return;
      }

      await DriveFileManager.initialize(accessToken);
      await DriveFileManager.updateFile(fileId, Buffer.from(content));

      res.status(200).json({
        status: 'success',
        message: 'File updated successfully',
      });
    } catch (error) {
      console.error('Update file error:', error);
      res.status(500).json({ status: 'error', message: 'Failed to update file' });
    }
  }
}
