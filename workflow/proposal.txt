const CueCardSchema = z.object({
  title: z.string(),

});

// aiService.ts
import { z } from "zod";
import OpenAI from "openai";
import { TopicSchema } from "../schemas/lessonSchema";

class AIService {
  private openai: OpenAI;

  constructor(openai: OpenAI) {
    this.openai = openai;
  }

  public async generateStructuredOutput(
    schema: z.ZodSchema,
    input: string,
    systemPrompt: string,
    schemaName: string
  ): Promise<any> {
    try {
      const response = await this.openai.beta.chat.completions.parse({
        model: "gpt-4o-mini",
        messages: [
          {
            role: "system",
            content: systemPrompt
          },
          {
            role: "user",
            content: input
          }
        ],
        max_tokens: 2000,
        response_format: zodResponseFormat(schema, schemaName)
      });

      return response.choices[0].message.parsed;
    } catch (error) {
      console.error("Error in generateStructuredOutput:", error);
      throw error;
    }
  }
}

// responseTranslator.ts
export function translateLessonToCueCards(
  lessonResponse: IStructuredLessonResponse, 
  nodeId: string
): CueCardResource[] {
  const cueCards: CueCardResource[] = [];

  // Add overview cue cards
  lessonResponse.overview.forEach(card => {
    cueCards.push({
      id: uuidv4(),
      title: card.title,
      nodeId,
      type: 'cueCard',
      content: [{ type: 'paragraph', children: [{ text: card.content }] }]
    });
  });

  // Function to process concept cue cards
  const processConceptCards = (concept: any) => {
    // Add concept content cards
    concept.lesson.content.forEach((card: any) => {
      cueCards.push({
        id: uuidv4(),
        title: card.title,
        nodeId,
        type: 'cueCard',
        content: [{ type: 'paragraph', children: [{ text: card.content }] }]
      });
    });

    // Add related concept cards if they exist
    if (concept.lesson.relatedConcept) {
      concept.lesson.relatedConcept.forEach((card: any) => {
        cueCards.push({
          id: uuidv4(),
          title: card.title,
          nodeId,
          type: 'cueCard',
          content: [{ type: 'paragraph', children: [{ text: card.content }] }]
        });
      });
    }

    // Process child concepts recursively
    concept.childConcepts.forEach((child: any) => {
      processConceptCards(child);
    });
  };

  // Process all main concepts
  lessonResponse.mainConcepts.forEach(concept => {
    processConceptCards(concept);
  });

  return cueCards;
}
