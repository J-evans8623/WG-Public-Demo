/**
 * authService.ts
 * 
 * Comprehensive service handling Google authentication, Drive operations,
 * automatic saving, and token management.
 * 
 * Responsibilities:
 * - Handles Google OAuth2 authentication flow and token refresh
 * - Manages access tokens and refresh tokens
 * - Provides Google Drive operations (create, read, update, delete)
 * - Handles file sharing and permissions
 * - Manages user persistence and JWT generation
 * - Handles automatic saving of documents
 * 
 * Dependencies:
 * - google-auth-library for OAuth2
 * - googleapis for Drive API
 * - Mongoose for user persistence
 * - JwtUtils for JWT operations
 */

import { OAuth2Client, Credentials } from 'google-auth-library';
import { drive_v3, google } from 'googleapis';
import { Readable } from 'stream';
import { debounce } from 'lodash';
import UserModel, { IUser } from '../models/userModel';
import JwtUtils from '../utils/jwtUtils';
import { RefreshTokenUtils } from '../utils/refreshTokenUtils';
import SessionStore from './sessionStore';


export interface DriveFile {
  id: string;
  name: string;
  mimeType: string;
  modifiedTime?: string;
}

export interface ShareSettings {
  role: 'reader' | 'writer' | 'commenter';
  type: 'user' | 'domain' | 'anyone';
  emailAddress?: string;
  domain?: string;
}

export interface AutoSaveConfig {
  enabled: boolean;
  interval: number;
  fileId?: string;
  userId?: string;
}
class AuthService {
  private static instance: AuthService;
  private googleClient: OAuth2Client;
  private drive: drive_v3.Drive | null = null;
  private credentials: Credentials | null = null;
  private tokenExpiryBuffer = 5 * 60 * 1000;

  private autoSaveConfig: AutoSaveConfig = {
    enabled: false,
    interval: 30000
  };
  private saveQueue: Map<string, Buffer> = new Map();
  private autoSaveTimer?: NodeJS.Timeout;
  private readonly debouncedSave: ReturnType<typeof debounce>;

  private constructor() {
    this.googleClient = new OAuth2Client(
      process.env.GOOGLE_CLIENT_ID,
      process.env.GOOGLE_CLIENT_SECRET,
      process.env.REDIRECT_URL
    );
    this.debouncedSave = debounce(this.processSaveQueue.bind(this), 2000);
  }

  static getInstance(): AuthService {
    if (!AuthService.instance) {
      AuthService.instance = new AuthService();
    }
    return AuthService.instance;
  }

  /**
   * Initializes the Drive client with provided access token
   * @param accessToken - Valid Google OAuth2 access token
   */
  async initializeDriveClient(accessToken: string): Promise<void> {
    try {
      await this.setCredentials({ access_token: accessToken });
      this.drive = google.drive({
        version: 'v3',
        auth: this.googleClient
      });
    } catch (error) {
      console.error('Failed to initialize Drive client:', error);
      throw error;
    }
  }

    /**
   * Sets OAuth2 credentials for the client.
   * @param credentials - OAuth2 credentials including access and refresh tokens.
   */
  private async setCredentials(credentials: Credentials): Promise<void> {
    this.credentials = credentials;
    this.googleClient.setCredentials(credentials);
    
    // Set up token refresh callback
    this.googleClient.on('tokens', (tokens) => {
      if (tokens.refresh_token) {
        this.credentials!.refresh_token = tokens.refresh_token;
      }
      if (tokens.access_token) {
        this.credentials!.access_token = tokens.access_token;
        this.credentials!.expiry_date = Date.now() + (tokens.expiry_date || 3600) * 1000;
      }
    });
  }

    /**
   * Ensures the access token is valid or refreshes it if necessary.
   * @param userId - ID of the user for retrieving the refresh token.
   */
  private async ensureValidToken(userId: string): Promise<void> {
    if (!this.credentials?.expiry_date) return;
    
    const now = Date.now();
    if (this.credentials.expiry_date - now <= this.tokenExpiryBuffer) {
      const refreshToken = await RefreshTokenUtils.getRefreshToken(userId);
      if (!refreshToken) throw new Error('No refresh token available');

      this.googleClient.setCredentials({ refresh_token: refreshToken });
      const { credentials } = await this.googleClient.refreshAccessToken();
      await this.setCredentials(credentials);
    }
  }
    /**
   * Revokes access tokens and clears stored credentials.
   * @param userId - ID of the user to revoke the refresh token.
   */
  async revokeAccess(userId: string): Promise<void> {
    await RefreshTokenUtils.revokeRefreshToken(userId);
    if (this.credentials?.access_token) {
      await this.googleClient.revokeToken(this.credentials.access_token);
    }
    this.credentials = null;
  }

    /**
   * Configures the auto-save mechanism for documents.
   * @param config - Partial configuration for auto-save.
   */
  configureAutoSave(config: Partial<AutoSaveConfig>): void {
    this.autoSaveConfig = { ...this.autoSaveConfig, ...config };
  
    if (this.autoSaveTimer) {
      clearInterval(this.autoSaveTimer);
    }
  
    if (this.autoSaveConfig.enabled && this.autoSaveConfig.fileId && this.autoSaveConfig.userId) {
      this.autoSaveTimer = setInterval(() => {
        this.processSaveQueue();
      }, this.autoSaveConfig.interval);
    }
  }
  

 
  /**
   * Queues content for auto-saving
   * @param content - Content to be saved
   */
  queueAutoSave(content: Buffer): void {
    if (!this.autoSaveConfig.fileId || !this.autoSaveConfig.userId) return;
    this.saveQueue.set(this.autoSaveConfig.fileId, content);
    this.debouncedSave();
  }
  

  /**
   * Processes the auto-save queue
   */
  private async processSaveQueue(): Promise<void> {
    if (!this.autoSaveConfig.fileId || !this.autoSaveConfig.userId) return;
  
    const content = this.saveQueue.get(this.autoSaveConfig.fileId);
    if (!content) return;
  
    try {
      const userId = this.autoSaveConfig.userId;
  
      await this.ensureValidToken(userId);
      await this.updateFile(this.autoSaveConfig.fileId, content, userId);
      this.saveQueue.delete(this.autoSaveConfig.fileId);
    } catch (error) {
      console.error('AutoSave failed:', error);
      // Keep in queue for retry
    }
  }
  

  /**
   * Verifies the validity of an access token
   * @param token - Access token to verify
   */
  private async verifyToken(token: string): Promise<void> {
    try {
      await this.googleClient.getTokenInfo(token);
    } catch (error) {
      console.error('Token verification failed:', error);
      throw new Error('Invalid or expired token');
    }
  }


  

  /**
   * Saves a file to Google Drive
   * @param fileName - Name of the file
   * @param content - File content as Buffer
   * @param folderId - Optional folder ID
   * @returns File ID of the saved file
   */
  async saveFile(fileName: string, content: Buffer, folderId?: string): Promise<string> {
    if (!this.drive) throw new Error('Drive client not initialized');

    try {
      const fileStream = new Readable();
      fileStream.push(content);
      fileStream.push(null);

      const requestBody = {
        name: `${fileName}.bloomdocx`,
        mimeType: 'application/vnd.bloom.document',
        ...(folderId && { parents: [folderId] }),
      };

      const response = await this.drive.files.create({
        requestBody,
        media: {
          mimeType: 'application/octet-stream',
          body: fileStream,
        },
        fields: 'id',
      });

      if (!response.data.id) {
        throw new Error('Failed to get file ID from response');
      }

      return response.data.id;
    } catch (error) {
      console.error('Error saving file to Drive:', error);
      throw error;
    }
  }

  async updateFile(fileId: string, content: Buffer, userId: string): Promise<void> {
    if (!this.drive) throw new Error('Drive client not initialized');
    
    await this.ensureValidToken(userId);
    
    try {
      const fileStream = new Readable();
      fileStream.push(content);
      fileStream.push(null);

      await this.drive.files.update({
        fileId,
        media: {
          mimeType: 'application/octet-stream',
          body: fileStream,
        },
      });
    } catch (error) {
      console.error('Error updating file:', error);
      throw error;
    }
  }
  /**
   * Retrieves a file from Google Drive
   * @param fileId - ID of the file to retrieve
   * @returns File content as Buffer
   */
  async getFile(fileId: string): Promise<Buffer> {
    if (!this.drive) throw new Error('Drive client not initialized');

    try {
      const response = await this.drive.files.get(
        { fileId, alt: 'media' },
        { responseType: 'arraybuffer' }
      );

      return Buffer.from(response.data as ArrayBuffer);
    } catch (error) {
      console.error('Error getting file:', error);
      throw error;
    }
  }

  /**
   * Lists all .bloomdocx files in Drive
   * @returns Array of DriveFile objects
   */
  async listFiles(): Promise<DriveFile[]> {
    if (!this.drive) throw new Error('Drive client not initialized');

    try {
      const response = await this.drive.files.list({
        q: "mimeType='application/vnd.bloom.document'",
        fields: 'files(id, name, mimeType, modifiedTime)',
      });

      return (response.data.files || [])
        .filter((file): file is Required<Pick<drive_v3.Schema$File, 'id' | 'name' | 'mimeType'>> & 
          { modifiedTime?: string } =>
          Boolean(file.id && file.name && file.mimeType)
        )
        .map(file => ({
          id: file.id!,
          name: file.name!,
          mimeType: file.mimeType!,
          modifiedTime: file.modifiedTime || undefined,
        }));
    } catch (error) {
      console.error('Error listing files:', error);
      throw error;
    }
  }

  /**
   * Shares a file with specified permissions
   * @param fileId - ID of the file to share
   * @param settings - Share settings including permissions
   */
  async shareFile(fileId: string, settings: ShareSettings): Promise<void> {
    if (!this.drive) throw new Error('Drive client not initialized');

    try {
      const requestBody = {
        role: settings.role,
        type: settings.type,
        ...(settings.emailAddress && { emailAddress: settings.emailAddress }),
        ...(settings.domain && { domain: settings.domain }),
      };

      await this.drive.permissions.create({
        fileId,
        requestBody,
        sendNotificationEmail: true,
      });
    } catch (error) {
      console.error('Error sharing file:', error);
      throw error;
    }
  }

  // Original AuthService methods

  generateAuthUrl(actionType: string): string {
    const sessionId = SessionStore.createSession();
    return this.googleClient.generateAuthUrl({
      access_type: 'offline',
      scope: [
        'https://www.googleapis.com/auth/userinfo.profile',
        'https://www.googleapis.com/auth/drive.file',
        'https://www.googleapis.com/auth/drive.appdata',
      ],
      prompt: 'consent',
      state: `${sessionId}:${actionType}`
    });
  }

  async getTokensFromCode(code: string): Promise<Credentials> {
    const { tokens } = await this.googleClient.getToken(code);
    return tokens;
  }

  async validateToken(token: string): Promise<any> {
    try {
      const response = await fetch(
        'https://www.googleapis.com/oauth2/v3/userinfo',
        {
          headers: { Authorization: `Bearer ${token}` },
        }
      );
      
      if (!response.ok) throw new Error('Failed to validate token');
      
      return await response.json();
    } catch (error) {
      throw new Error('Invalid Google OAuth token.');
    }
  }

  async retrieveUser(googleUserData: any): Promise<IUser> {
    const { sub: googleId, email } = googleUserData;
    let user = await UserModel.findOne({ googleId });
    if (!user) {
      user = await UserModel.create({ googleId, email });
    }
    return user;
  }

  generateJWT(user: IUser): string {
    return JwtUtils.generateJWT({
      id: user._id,
      email: user.email,
      accountType: user.accountType,
    });
  }
}

export default AuthService.getInstance();
