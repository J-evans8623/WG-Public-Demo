/**
 * src/services/DriveFileManager.ts
 * 
 * Manages Google Drive file operations and coordinates with local file operations.
 * Acts as a facade for DriveService, DriveShareService, and AutoSaveService.
 * 
 * Dependencies:
 * - DriveService for basic Drive operations
 * - DriveShareService for sharing functionality
 * - AutoSaveService for automatic saving
 * - documentUtils for file serialization
 */

import DriveService, { DriveFile } from './driveService';
import DriveShareService from './driveShareService';
import AutoSaveService, { AutoSaveConfig } from './autoSaveService';
import { ShareSettings } from './driveShareService';

export class DriveFileManager {
  private static instance: DriveFileManager;

  private constructor() {}

  /**
   * Returns the singleton instance of DriveFileManager
   */
  public static getInstance(): DriveFileManager {
    if (!DriveFileManager.instance) {
      DriveFileManager.instance = new DriveFileManager();
    }
    return DriveFileManager.instance;
  }

  /**
   * Initializes the Drive client with the provided access token
   * @param accessToken - Google OAuth access token
   */
  async initialize(accessToken: string): Promise<void> {
    await DriveService.initializeDriveClient(accessToken);
  }

  /**
   * Saves content to Google Drive
   * @param fileName - Name of the file
   * @param content - Serialized document content
   * @param folderId - Optional folder ID to save the file in
   * @returns Promise resolving to the file ID of the saved file
   */
  async saveToGoogleDrive(
    fileName: string,
    content: Buffer,
    folderId?: string
  ): Promise<string> {
    try {
      const fileId = await DriveService.saveFile(fileName, content, folderId);
      return fileId;
    } catch (error) {
      console.error('Error saving to Google Drive:', error);
      console.error('Error details:', JSON.stringify(error, null, 2));
      throw new Error('Failed to save file to Google Drive');
    }
  }

  /**
   * Loads a file from Google Drive
   * @param fileId - ID of the file to load
   * @returns Promise resolving to the file content
   */
  async loadFromGoogleDrive(fileId: string): Promise<Buffer> {
    try {
      const content = await DriveService.getFile(fileId);
      return content;
    } catch (error) {
      console.error('Error loading from Google Drive:', error);
      throw new Error('Failed to load file from Google Drive');
    }
  }

  /**
   * Lists available .bloomdocx files in Google Drive
   * @returns Promise resolving to an array of DriveFile objects
   */
  async listDriveFiles(): Promise<DriveFile[]> {
    try {
      const files = await DriveService.listFiles();
      return files;
    } catch (error) {
      console.error('Error listing Drive files:', error);
      throw new Error('Failed to list Drive files');
    }
  }

  /**
   * Configures autosave functionality
   * @param config - AutoSave configuration options
   */
  configureAutosave(config: AutoSaveConfig): void {
    AutoSaveService.configure(config);
  }

  /**
   * Shares a file with specified settings
   * @param fileId - ID of the file to share
   * @param settings - Share settings including permissions and target users
   */
  async shareFile(fileId: string, settings: ShareSettings): Promise<void> {
    try {
      await DriveShareService.shareFile(fileId, settings);
    } catch (error) {
      console.error('Error sharing file:', error);
      throw new Error('Failed to share file');
    }
  }

  /**
   * Updates an existing file in Google Drive
   * @param fileId - ID of the file to update
   * @param content - New content for the file
   */
  async updateFile(fileId: string, content: Buffer): Promise<void> {
    try {
      await DriveService.updateFile(fileId, content);
    } catch (error) {
      console.error('Error updating file:', error);
      throw new Error('Failed to update file');
    }
  }
}

export default DriveFileManager.getInstance();

/**
 * authRoutes.ts
 * 
 * Provides authentication-related API routes using Express Router.
 * Responsibilities:
 * - Defines routes for user authentication via Google OAuth2.
 * - Implements a silent callback endpoint (`/oauth/callback`) for OAuth2 token exchanges.
 * - Provides a login route for direct user authentication.
 * - Communicates tokens securely to the parent app using `postMessage`.
 * Dependencies:
 * - Express for routing.
 * - AuthService for handling token exchanges and Google OAuth2 logic.
 * - AuthController for handling authentication requests.
 * - RateLimiter to prevent API abuse.
 */

import express from 'express';
import AuthService from '../services/authService';
import AuthController from '../controllers/authController';
import RateLimiter from '../middleware/rateLimiter';

const router = express.Router();

/**
 * POST /api/login
 * 
 * Authenticates a user using a Google OAuth token and returns a JWT.
 * Rate-limited to prevent abuse.
 */
router.post('/login', RateLimiter, AuthController.login);

/**
 * GET /oauth/callback
 * 
 * Silent OAuth2 callback endpoint.
 * Handles the token exchange after Google redirects the user here and uses `postMessage`
 * to securely send tokens and state back to the parent application.
 * 
 * Query Parameters:
 * - code: Authorization code from Google.
 * - state: Encoded user intent (e.g., `file-sharing`, `premium-service`).
 * 
 * Responses:
 * - Sends an HTML script to post the tokens and state back to the parent window and closes the popup.
 */
router.get('/oauth/callback', async (req, res) => {
  const { code, state } = req.query;

  try {
    if (!code) {
      throw new Error('Authorization code is missing.');
    }

    // Exchange the authorization code for tokens
    const tokens = await AuthService.getTokensFromCode(code as string);

    // Generate the HTML response to send tokens and state back to the parent app
    const responseHtml = `
      <script>
        // Post tokens and state back to the parent window
        window.opener.postMessage(
          ${JSON.stringify({ tokens, state })},
          window.opener.location.origin
        );
        // Close the popup
        window.close();
      </script>
    `;
    res.send(responseHtml);
  } catch (error) {
    const errorMessage = error instanceof Error ? error.message : 'Unknown error occurred';
    console.error('OAuth2 callback error:', errorMessage);
    res.status(400).send('Authentication failed.');
  }
});

export default router;

/**
 * Saves document to Google Drive
 * @param nodes - Map of concept nodes
 * @param colorTheme - Current color theme
 * @param primaryColor - Primary color theme
 * @param backgroundColor - Background color theme
 * @param fileName - Name of the file
 * @param cueCards - Map of cue card resources
 * @param subtrees - Map of subtree data
 * @param pictures - Map of picture resources
 * @param practiceSheets - Map of practice sheet resources
 * @param driveFileManager - Instance of DriveFileManager
 * @returns Promise resolving to the Google Drive file ID
 */
export const saveToGoogleDrive = async (
  nodes: Map<string, ConceptNode>,
  colorTheme: ColorTheme,
  primaryColor: string,
  backgroundColor: string,
  fileName: string,
  cueCards: Map<string, ResourceCard[]>,
  subtrees: Map<string, SubtreeData>,
  pictures: Map<string, ResourceCard[]>,
  practiceSheets: Map<string, ResourceCard[]>,
  driveFileManager: typeof DriveFileManager
): Promise<string> => {
  const zip = new JSZip();

  // Reuse existing serialization logic
  const serializedDocument = serializeDocument(
    nodes,
    colorTheme,
    primaryColor,
    backgroundColor,
    fileName
  );
  const documentJson = JSON.stringify(serializedDocument);
  zip.file('document/document.json', documentJson);

  // Add other files using existing serialization methods
  const serializedCueCards = serializeCueCards(cueCards as Map<string, CueCardResource[]>);
  zip.file('document/cuecards.json', JSON.stringify(serializedCueCards));

  const serializedSubtrees = serializeSubtrees(subtrees);
  zip.file('document/subtrees.json', JSON.stringify(serializedSubtrees));

  const serializedPictures = serializePictures(pictures as Map<string, PictureResource[]>);
  zip.file('document/pictures.json', JSON.stringify(serializedPictures));

  const serializedPracticeSheets = serializePracticeSheets(
    practiceSheets as Map<string, PracticeSheetResource[]>
  );
  zip.file('document/practiceSheets.json', JSON.stringify(serializedPracticeSheets));

  // Add images to the ZIP
  for (const [nodeId, pictureArray] of pictures.entries()) {
    for (const picture of pictureArray as PictureResource[]) {
      if (picture.image) {
        try {
          const imageData = await picture.image.arrayBuffer();
          zip.file(`document/images/${picture.id}_${picture.image.name}`, imageData);
        } catch (error) {
          console.error(`Error saving image ${picture.id}:`, error);
        }
      }
    }
  }

  // Generate ZIP content as Buffer
  const content = await zip.generateAsync({ type: 'nodebuffer' });

/**
 * authController.ts
 * 
 * Handles API requests related to authentication, such as user login.
 */

import { Request, Response } from 'express';
import AuthService from '../services/authService';

/**
 * AuthController
 * 
 * Contains methods to handle authentication-related HTTP requests.
 */
class AuthController {
  /**
   * Handles user login via Google OAuth token.
   * 
   * @param req - Express Request object containing the Google OAuth token.
   * @param res - Express Response object to send back the JWT and user details.
   */
  public async login(req: Request, res: Response): Promise<void> {
    try {
      const { token } = req.body;
      if (!token) {
        res.status(400).json({ status: 'error', message: 'Token is required.' });
        return;
      }

      // Validate the Google OAuth token
      const googleUserData = await AuthService.validateToken(token);
      if (!googleUserData) {
        res.status(401).json({ status: 'error', message: 'Invalid token.' });
        return;
      }

      // Retrieve or create the user
      const user = await AuthService.retrieveUser(googleUserData);

      // Generate JWT
      const jwtToken = AuthService.generateJWT(user);

      res.status(200).json({
        status: 'success',
        token: jwtToken,
        user: {
          email: user.email,
          accountType: user.accountType,
        },
      });
    } catch (error) {
      console.error('Login error:', error);
      res.status(500).json({ status: 'error', message: 'Internal server error.' });
    }
  }
}

export default new AuthController();
