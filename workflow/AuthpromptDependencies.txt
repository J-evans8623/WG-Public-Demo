Please note that :


  'HandleSign in Success' is called via 'onSignInSuccess' prop of AuthpromptModal

 
  const handleSignInSuccess = async (tokenResponse: any) => {
    try {
      const response = await fetch(`${apiServerUrl}/api/login`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ 
          token: tokenResponse.access_token  // Send access_token instead of credential
        }),
      });
      
      const data = await response.json();
      if (data.status === 'success') {
        signIn(data.token);
        setIsAuthModalOpen(false);
      } else {
        alert(data.message || 'Authentication failed.');
      }
    } catch (error) {
      console.error('Login Error:', error);
      alert('An error occurred during authentication.');
    }
  };
